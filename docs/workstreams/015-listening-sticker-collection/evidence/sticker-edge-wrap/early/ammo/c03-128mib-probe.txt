import {actualShell} from '/tmp/webpod-fermi-shell';
import {Vector3} from '/Users/vinicius/code/webPod/packages/device/node_modules/three';
const phase=(name)=>console.error(JSON.stringify({phase:name,at:performance.now()}));
const dir='/tmp/webpod-ammo-79190a1f';
const data=await Bun.file(`${dir}/seed-preflight-tolerant.json`).json();

const factory=require(`${dir}/ammo.wasm.js`),init=performance.now();
const Ammo=await factory({INITIAL_MEMORY:Number(process.env.MEMORY_MIB??64)*1024*1024,wasmBinary:new Uint8Array(await Bun.file(`${dir}/ammo.wasm.wasm`).arrayBuffer())});

for(const hz of[Number(process.env.RUN_HZ??120)]){
 const ledger=[],own=(name,object)=>{ledger.push({name,object,destroyed:false});return object;};

  phase('start native world');const constructionStart=performance.now();

  // Candidate-support copies duplicate original front geometry and are not added twice.

  phase({name:'finished clusters',staticTriangles,clusterCount,bendingLinks,triangleCopyMs,nativeBvhMs,clusterMs});world.addSoftBody(soft,1,-1);addedSoft=true;if(process.env.CONSTRUCTION_ONLY==='1')throw Error('Construction-only diagnostic complete');

   for(let i=0;i<1089;i++){if(pins.has(i))continue;const node=nodes.at(i),p=node.get_m_x(),v=node.get_m_v(),target=data.targets[i].position;const dx=target[0]-p.x(),dy=target[1]-p.y(),dz=target[2]-p.z();
    if(t>=press&&Math.hypot(dx,dy,dz)>.25)continue;

   if(completed===Math.round(press*hz)||completed===steps){const c=performance.now();snapshots.push({phase:completed===steps?'fixed-target capture end':'press end',time:completed*dt,nodes:capture()});copyMs+=performance.now()-c;}
  }
  const c=performance.now(),finalNodes=capture();copyMs+=performance.now()-c;
  result={hz,status,requestedSteps:steps,completedSteps:completed,physicalTime:completed*dt,pressSeconds:press,captureSeconds:.5,forceMs,nativeMs,solverPlusForceMs:forceMs+nativeMs,copyMs,triangleCopyMs,nativeBvhMs,constraintMs,clusterMs,constructionMs,staticTriangles,bendingLinks,clusterCount,collidableClusters:clusterCount-1,clusterQualification:'Pinned implementation includes a non-colliding aggregate cluster; count minus one assumes exactly that master remains.',heapAfterConstruction,rest,indices,finalNodes,snapshots,times,pins:data.pins};

 }
 result.cleanup=ledger.map(({name,destroyed})=>({name,destroyed}));result.heapAfterCleanup=Ammo.HEAP8.byteLength;results.push(result);
 await Bun.write(`${dir}/c03-fixed-target-${hz}.json`,JSON.stringify(result));
 console.log(JSON.stringify(Object.fromEntries(Object.entries(result).filter(([k])=>!['rest','indices','finalNodes','snapshots','times','cleanup'].includes(k)))));
}
shell.dispose();await Bun.write(`${dir}/c03-fixed-target-summary.json`,JSON.stringify({variant:'LIMITED frozen initial exterior target substrate/press diagnostic.
